{{- /* Resolve a story/theme image object to a locally stored, optimized resource. Returns false unless the image pipeline has stored it under assets/ (image.local). Remote URLs are never hotlinked. Result: {src, permalink, width, height, srcset, vector, alt, credit, sourceUrl}. Raster masters are at most 600px; Hugo adds smaller WebP widths for srcset (never upscaled). SVGs are served as-is (already sanitized) with dimensions from the JSON. */ -}} {{- $out := false -}} {{- with .image -}} {{- $image := . -}} {{- with $res := and .local (resources.Get .local) -}} {{- if eq $res.MediaType.SubType "svg" -}} {{- $out = dict "src" $res.RelPermalink "permalink" $res.Permalink "width" ($image.width | default 0) "height" ($image.height | default 0) "srcset" "" "vector" true -}} {{- else -}} {{- $srcset := slice -}} {{- range $w := site.Params.imageWidths | default (slice 320 480) -}} {{- if gt $res.Width (add $w 80) -}} {{- with $res.Process (printf "resize %dx webp" $w) -}} {{- $srcset = $srcset | append (printf "%s %dw" .RelPermalink .Width) -}} {{- end -}} {{- end -}} {{- end -}} {{- $srcset = $srcset | append (printf "%s %dw" $res.RelPermalink $res.Width) -}} {{- $out = dict "src" $res.RelPermalink "permalink" $res.Permalink "width" $res.Width "height" $res.Height "srcset" (delimit $srcset ", ") "vector" false -}} {{- end -}} {{- end -}} {{- if $out -}} {{- $out = merge $out (dict "alt" (.alt | default "") "credit" (.credit | default "") "sourceUrl" (.sourceUrl | default "")) -}} {{- end -}} {{- end -}} {{- return $out -}}
